cmake_minimum_required(VERSION 3.16)
project(ZcAiLibConsumer LANGUAGES CXX)
find_package(ZcAiLib 0.2 CONFIG REQUIRED PATHS "${SDK_PREFIX}" NO_DEFAULT_PATH)
enable_testing()
foreach(kind IN ITEMS ZcAiLib ZcAiLibStatic)
    if(TARGET ZcAiLib::${kind})
        add_executable(consumer_${kind} main.cpp)
        target_link_libraries(consumer_${kind} PRIVATE ZcAiLib::${kind})
        add_test(NAME consumer_${kind} COMMAND consumer_${kind})
    endif()
endforeach()
